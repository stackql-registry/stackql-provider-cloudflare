REPLACE cloudflare.api_gateway.managed_resources_operation

WHERE 
zone_id = '{{ zone_id }}' --required
AND name = '{{ name }}' --required
AND selector = '{{ selector }}' --required
RETURNING
errors,
messages,
result,
success;
